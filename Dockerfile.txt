# =========================================================
# BUILD STAGE
# =========================================================

FROM mcr.microsoft.com/dotnet/sdk:8.0 AS build

WORKDIR /src

COPY ["CyberChatDemo.csproj", "./"]

RUN dotnet restore "CyberChatDemo.csproj"

COPY . .

RUN dotnet publish "CyberChatDemo.csproj" \
    -c Release \
    -o /app/publish \
    /p:UseAppHost=false


# =========================================================
# RUNTIME STAGE
# =========================================================

FROM mcr.microsoft.com/dotnet/aspnet:8.0 AS final

WORKDIR /app

COPY --from=build /app/publish .

ENV ASPNETCORE_ENVIRONMENT=Production

EXPOSE 10000

ENTRYPOINT ["sh", "-c", "dotnet CyberChatDemo.dll --urls http://0.0.0.0:${PORT:-10000}"]
